  const preferredOrder: Side[] = (() => {
    switch (step.position) {
      case "top":    return ["top",    "left", "right", "bottom"]
      case "bottom": return ["bottom", "left", "right", "top"]
      case "left":   return ["left",   "top",  "bottom", "right"]
      case "right":  return ["right",  "top",  "bottom", "left"]
      default:       return ["bottom", "left", "right", "top"]
    }
  })()

  function rawForSide(side: Side, sr: SpotlightRect): { top: number; left: number } {

          left: Math.max(16, Math.min(sr.left - 8, viewportWidth - BUBBLE_WIDTH - 16)),

  function overlaps(pos: { top: number; left: number }, sr: SpotlightRect): boolean {
    const bT = pos.top,  bB = pos.top  + BUBBLE_HEIGHT_ESTIMATE
